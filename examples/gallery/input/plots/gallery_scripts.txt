# == A Distributions

# A1 Histogram, step outline
hist --datafile Showers -x Response --hue Particle --select Particle -s electron pion --bins 60 --rangex 0.4 1.4 --flow hint --labelx 'Response $E_{reco}/E_{true}$' --title 'Calorimeter response to electrons and pions' --output_suffix A1

# A3 Filled / stacked histogram (stacked MC)
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution Sample -s Resonance Background Signal --stacked --plot_type fill --labelx 'Mass (GeV)' --labely 'Events / 2 GeV' --title 'Stacked simulation as filled steps' --output_suffix A3

# A4 Normalised / density histogram
hist --datafile Showers -x Hits --hue Particle --bins 50 --density --logy --labelx 'Hits' --title 'Hit multiplicity, normalised to unit area' --output_suffix A4

# A5 Weighted histogram
hist --datafile Showers -x TrueEnergy --hue Particle --select Particle -s electron pion --weight Weight --bins 38 --rangex 5 100 --labelx 'Beam energy (GeV)' --labely 'Events (prescale-weighted)' --title 'Beam energy, weighted by the trigger prescale' --output_suffix A5

# A6 G2 Histogram with Poisson (Garwood) errors
hist --datafile Showers -x RecoEnergy --select Particle -s muon --bins 40 --rangex 0 6 --errory_type poisson --labelx 'Deposited energy (GeV)' --title 'Muon deposits with Poisson intervals' --output_suffix A6

# A7 KDE
hist --datafile Scores -x Score --hue Class --select Model -s BDT --hist_type kde --labelx 'BDT score' --title 'Score densities' --output_suffix A7

# A8 ECDF / cumulative

# A9 Box plot per category

# A10 Violin per bin
line --datafile Showers -x TrueEnergy -y Residual --binned --plot_type violin --select Particle -s pion --bins 8 --rangex 5 75 --labelx 'Beam energy (GeV)' --labely '$(E_{reco}-E_{true})/E_{true}$' --title 'Pion energy residuals per beam-energy bin' --output_suffix A10

# A10b Violin / strip / box per category
cat --datafile Showers -x Particle -y Response --kind violin --labely 'Response' --title 'Response per particle' --output_suffix A10b

# A11 Ridgeline (joy plot)
hist --datafile Scores -x Score --hue Model --select Class -s Background --hist_type kde --ridge --rangex 0 1 --labelx 'Background score' --title 'Background scores of three classifiers' --output_suffix A11

# A12 Rug
hist --datafile Hubble1929 -x Distance --select Sample -s 'Individual nebulae' --hist_type kde --rug --rangex 0 2.5 --labelx 'Distance (Mpc)' --title "Distances of Hubble's 24 nebulae" --output_suffix A12
