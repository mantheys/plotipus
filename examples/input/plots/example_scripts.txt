# line: one series per --hue value, line style per --style value
line --datafile Spectra -x Energy -y Counts --hue Sample --style Stage --iterable_mapping sample_names --iterable_color_mapping sample_colors --plot_type step --logy --labelx 'Energy (MeV)' --title 'Spectra by Stage'
# line with a lower panel: ratio to the first series, and a stored column
line --datafile Spectra -x Energy -y Counts --hue Sample --select Stage -s Raw --plot_type step --logy --lower ratio --lower_labely 'Ratio to Signal' --output_suffix ratio
line --datafile Spectra -x Energy -y Counts --hue Sample --select Stage -s Raw --stacked --logy --lower group --reference_group signal --other_group background calibration --group_significance --lower_labely '$S/\sqrt{B}$' --lower_plot_type stepmid
# line, one series per config/name file, plus their sum
line --datafile Efficiency --configs small large --names signal -x Values -y Efficiency --by_config --plot_type line --errory --errory_type bands --combine_op mean --combined_label 'Average'
# line, binned estimator over raw per-event arrays
line --datafile Events -x TrueEnergy -y Resolution --hue Sample --binned --estimator mean --bins 15 --plot_type errorbar --labely 'Mean Resolution'
# hist: 1D histograms of raw arrays
hist --datafile Events -x RecoEnergy --hue Sample --bins 40 --labely Density --labelx 'Reco Energy (MeV)'
# map: 2D histogram and contours of a chi2 grid
map --datafile Events -x TrueEnergy -y RecoEnergy --iterable Sample --diagonal --logz
map --datafile Contour -x XGrid -y YGrid -z ZGrid --contour --contour_sigmas 1 2 3 --fill_contours
# cat: one value per category
cat --datafile Spectra -x Stage -y Counts --integrate_over Energy --hue Sample --kind bar --labely 'Total Counts'
# scatter: points with size and colour encodings
scatter --datafile Events -x TrueEnergy -y Hits --colorby RecoEnergy --select Sample -s signal --marker_size 4
