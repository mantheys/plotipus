# The gallery's tables (catalogue section J), same format as
# ../plots/gallery_scripts.txt: plotipus run gallery --tables.

# == J Tables

# J1 Values ± errors per config and variable
table --datafile Summary -y Value --variables Efficiency Purity Rate --title 'Detector performance' --output_name J1.tex

# J2 Efficiency table with Clopper–Pearson intervals
table --datafile TriggerEfficiency -x Energy --rangex 30 100 -y Efficiency --variable_name Particle --variables electron pion --clopper_pearson Passed --title 'Trigger efficiency (\%) above 30 GeV' --output_name J2.tex

# J3 siunitx / booktabs output with caption and label
table --datafile Summary -y Value --variables Efficiency Purity Rate --siunitx --caption 'Efficiency, purity and rate of the two layouts.' --label tab:summary --output_name J3.tex

# J4 HTML twin of a table
table --datafile Summary -y Value --variables Efficiency Purity Rate --html --title 'Detector performance' --output_name J4.tex

# J5 Cut flow
table --datafile Yields -y Counts --hue Particle --variable_name Stage --cutflow --output_name J5.tex

# J6 Summary statistics table (mean, variance, correlation, line)
table --datafile Anscombe -x X -y Y --select Type -s Data --row_name Dataset --stats mean var corr line --title "Anscombe's quartet: the same statistics" --output_name J6.tex

# J7 Fit-parameter table from stored fits
table --datafile Moseley -x AtomicNumber -y SqrtFrequency --select Type -s Data --fit_params --row_name Series --title "Moseley's law fits" --output_name J7.tex
