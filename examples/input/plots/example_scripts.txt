# The example figures, also those of docs/options.md (make -C examples docs).
# Data: input/data/Distributions.pkl and the files derived from it (make_data.py).

# Shared options: panels, encodings, palettes, errors, reference lines
line --datafile Distributions -x Bin -y Counts --hue Sample --variable_column Distribution --variables Gaussian Poisson Exponential --plot_type step --labelx x --title 'Prebinned counts, one panel per distribution' --output_suffix facets
line --datafile Distributions -x Bin -y Counts --hue Distribution --style Sample --select Distribution -s Exponential Landau Chi-squared --plot_type step --labelx x --title 'Colour by --hue, line style by --style' --output_suffix encodings
line --datafile Distributions -x Bin -y Counts --hue Distribution --select Sample Distribution -s Data Exponential Landau Chi-squared Log-normal --plot_type step --palette okabe-ito --labelx x --title 'The okabe-ito palette' --output_suffix palette
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution -s Gaussian --plot_type errorbar --errory --labelx x --title 'Error bars from the CountsError column' --output_suffix errors
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution -s Exponential --plot_type step --errory --errory_type bands --logy --labelx x --title 'Error bands on a log scale' --output_suffix bands
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution -s Breit-Wigner --plot_type step --vertical 89.94 92.44 --horizontal 128 --note 'FWHM = 2.50 GeV' --labelx 'Mass (GeV)' --title 'Reference lines and a note' --output_suffix overlays
line --datafile Histogram --configs data simulation --names landau -x Bin -y Counts --by_config --data_hue data --errory --labelx x --title 'One file per config' --output_suffix configs

# Histograms of samples
hist --datafile Distributions -x Value --hue Sample --weight Weight --select Distribution -s Breit-Wigner --bins 40 --rangex 81 101 --flow show --labelx 'Mass (GeV)' --title 'Histogram of samples with under/overflow' --output_suffix flow

# Lower panels
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution -s Landau --plot_type step --lower ratio --reference_value Simulation --labelx x --lower_labely 'Data / Sim.' --title 'Ratio to a reference series' --output_suffix ratio
line --datafile Distributions -x Bin -y Counts --select Distribution Sample -s Poisson Data --plot_type step --lower column:Pull --labelx k --lower_labely Pull --title 'Lower panel from a stored column' --output_suffix column
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution Sample -s Resonance Background Signal --stacked --lower group --reference_group Signal --other_group Background --group_significance --lower_labely '$S/\sqrt{B}$' --labelx 'Mass (GeV)' --labely 'Events / 2 GeV' --title 'Stacked simulation and the signal significance' --output_suffix significance

# Data and simulation
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution -s Landau --data_hue Data --errory --lower ratio --labelx x --lower_labely 'Data / Sim.' --title 'Data against simulation' --output_suffix datamc
line --datafile Distributions -x Bin -y Counts --hue Sample --select Distribution -s Resonance --data_hue Data --stacked --errory --lower ratio --labelx 'Mass (GeV)' --labely 'Events / 2 GeV' --lower_labely 'Data / Sim.' --title 'Data against stacked simulation' --output_suffix datastack

# Fits
line --datafile Distributions -x Bin -y Counts --select Distribution Sample -s Gaussian Data --plot_type errorbar --errory --fit --lower pull --labelx x --title 'Stored fit with pulls' --output_suffix fit
line --datafile Distributions -x Bin -y Counts --select Distribution Sample -s Resonance Data --fit --data_hue Data --lower pull --chi2 --max_fit_params 3 --labelx 'Mass (GeV)' --labely 'Events / 2 GeV' --title 'Data with a signal + background fit' --output_suffix datafit

# 2D: histograms, colorbars, sigma regions, contours of a map
map --datafile Distributions -x Value -y Smeared --select Distribution -s Exponential --iterable Sample --diagonal --logz --labelx True --labely Measured --title '2D histograms with a shared colorbar' --output_suffix colorbar
map --datafile Distributions -x X -y Y --select Distribution -s 'Bivariate Gaussian' --iterable Sample --contour --bins 30 --logz --title 'A bivariate Gaussian and its 1, 2 and 3 sigma regions' --output_suffix sigmas
map --datafile Distributions -x MeanScan -y SigmaScan -z Chi2Scan --select Distribution -s Gaussian --iterable Sample --contour --fill_contours --labelx '$\mu$' --labely '$\sigma$' --title 'Confidence regions of a $\Delta\chi^2$ scan' --output_suffix contours

# Binned estimators
line --datafile Distributions -x Value -y Residual --binned --estimator std --hue Sample --select Distribution -s Exponential --bins 11 --rangex 0 10 --labelx True --labely 'Measured - True' --title 'Binned estimator: resolution' --output_suffix binned

# Other kinds: categories and scatter
cat --datafile Distributions -x Distribution --hue Sample -y Counts --select Distribution -s Resonance --kind bar --labely Events --title 'Yields: summed counts per sample' --output_suffix yields
scatter --datafile Distributions -x Value -y Smeared --colorby Residual --select Distribution Sample -s Exponential Data --marker_size 4 --labelx True --labely Measured --title 'Points coloured by a third column' --output_suffix scatter
